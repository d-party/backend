44def _logical_delete_room_if_empty(room_id_str: str) -> bool: 

45 """Delete room only if it is still empty. Returns True if deleted.""" 

46 qs = AnimeRoom.objects.alive().filter(room_id=room_id_str) 

47 if not qs.exists(): 

48 return False 

49 if AnimeUser.objects.alive().filter(room_id=room_id_str).exists(): 

50 return False 

51 # ルームが終了するので、リアクションを集計テーブルへ畳んで生データを削除する。 

52 fold_room_reactions(room_id_str) 

53 qs.delete() # logical 

54 return True

57def _cancel_pending_room_delete(room_id_str: str) -> None: 

58 """Cancel a scheduled grace-period delete if any (e.g., on rejoin).""" 

59 task = _pending_room_deletes.pop(room_id_str, None) 

60 if task and not task.done(): 

61 task.cancel()

64def _schedule_room_delete(room_id_str: str, delay: float = ROOM_GRACE_SECONDS) -> None: 

65 """Schedule a logical-delete after `delay` seconds, replacing any prior task.""" 

66 _cancel_pending_room_delete(room_id_str)

68 async def _delete_after() -> None: 

69 try: 

70 await asyncio.sleep(delay) 

71 await _logical_delete_room_if_empty(room_id_str) 

72 except asyncio.CancelledError: 

73 pass 

74 finally: 

75 _pending_room_deletes.pop(room_id_str, None) 

76 

77 _pending_room_deletes[room_id_str] = asyncio.create_task(_delete_after())

542 async def leave_party(self): 

543 """サーバーから離脱する場合の共通処理 

544 データベースからの論理削除などを行い、ルーム内のユーザーに通知する 

545 """ 

546 if self.anime_room is None or self.anime_user is None: 

547 return 

548 leave = Leave(user=User(**self.anime_user.__dict__).model_dump()) 

549 response_data = GroupSend( 

550 response=leave, 

551 sender_channel_name=self.channel_name, 

552 ) 

553 

554 await self.database_delete_user() 

555 # 在室人数は AnimeUser から都度数える（AnimeRoom には人数カラムを持たない）。 

556 user_count = await self.database_get_user_count() 

557 # オーナー退室時自動削除。オーナーが抜けたら残りの参加者ごとルームを即削除し、 

558 # 全員へ room_deleted を通知する。猶予削除やホスト委譲は行わない。 

559 # （一方通行モード単体では自動削除しない。owner_leave_delete が有効なときだけ。） 

560 if self.anime_user.is_host and self._effective_owner_leave_delete(): 

561 room_id_str = str(self.anime_room.room_id) 

562 _cancel_pending_room_delete(room_id_str) 

563 server_message = ServerMessage(message_type="room_deleted") 

564 send_data = RoomSend( 

565 response=server_message, 

566 sender_channel_name=self.channel_name, 

567 ) 

568 await self.channel_layer.group_send( 

569 room_id_str, 

570 send_data.model_dump(mode="json"), 

571 ) 

572 await self.database_delete_room_and_users() 

573 await self.channel_layer.group_discard(room_id_str, self.channel_name) 

574 return 

575 if user_count < 1: 

576 # 即消しだとホストの一瞬切断・タブリロードでルームが消え、 

577 # ゲストが全員 failed_join になる。猶予期間中に誰かが再参加したら join 側で 

578 # cancel される。 

579 _schedule_room_delete(str(self.anime_room.room_id)) 

580 if user_count >= 1 and self.anime_user.is_host: 

581 next_host = await self.database_promote_next_host() 

582 if next_host is not None: 

583 server_message = ServerMessage(message_type="host_change") 

584 send_data = HostSend( 

585 response=server_message, sender_channel_name=self.channel_name 

586 ) 

587 await self.channel_layer.group_send( 

588 str(self.anime_room.room_id), 

589 send_data.model_dump(mode="json"), 

590 ) 

591 await self.channel_layer.group_send( 

592 str(self.anime_room.room_id), 

593 response_data.model_dump(mode="json"), 

594 ) 

595 user_list = await self.database_user_list() 

596 user_list_data = UserList(user_list=user_list) 

597 response_data = RoomSend( 

598 response=user_list_data, 

599 sender_channel_name=self.channel_name, 

600 ) 

601 await self.channel_layer.group_send( 

602 str(self.anime_room.room_id), 

603 response_data.model_dump(mode="json"), 

604 ) 

605 await self.channel_layer.group_discard( 

606 str(self.anime_room.room_id), self.channel_name 

607 )

680 @database_sync_to_async 

681 def database_promote_next_host(self): 

682 """残った生存ユーザーのうち最古の 1 人をホストへアトミックに昇格させる。

684 次ホストの選定と昇格を 1 トランザクション（``select_for_update``）で行う。

692 .alive() 

693 .filter(room_id=self.anime_room.room_id) 

694 .order_by("created_at") 

695 .first() 

696 )
